    // Меняем основной текст

      button.innerText = 'Перевод с малышкового';

    if (!isSecondTitle) {
      titleEl.innerHTML = titles[1];
    } else {
      titleEl.innerHTML = titles[0];
    }
    isSecondTitle = !isSecondTitle;
  });
